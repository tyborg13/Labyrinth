"""Verbose, fresh-profile replays to attribute shutdown-only ObjectDB warnings."""
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
import importlib.util,json,os,subprocess,sys,time
PROJECT=Path('/Users/borgerding/workspace/Labyrinth.worktrees/qa-30')
EVIDENCE=Path('/private/tmp/labyrinth_codex/qa-30')
REPLAY_PROCESSES=2
CONTROL_SEED=2

def execute(seed):
    directory=EVIDENCE/('shutdown-replay-'+str(seed));profile=directory/f'seed-{seed:06d}'
    profile.mkdir(parents=True)
    command=[sys.executable,'tools/godot_task_runner.py','--task-id','qa-30','--timeout','4500','--stream','--','godot','--verbose','--headless','--path','.','--script','tools/run_hunt.gd','--','--seed',str(seed),'--output-dir',str(profile)]
    (profile/'command.json').write_text(json.dumps(dict(cwd=str(PROJECT),env=dict(LABYRINTH_CEL_LOOK='off'),argv=command),indent=2)+'\n')
    started=time.time()
    with (profile/'godot.log').open('w') as log:
        result=subprocess.run(command,cwd=PROJECT,env=dict(os.environ,LABYRINTH_CEL_LOOK='off'),stdout=log,stderr=subprocess.STDOUT)
    spec=importlib.util.spec_from_file_location('qa30_shutdown_hunt',PROJECT/'tools/run_hunt.py');hunt=importlib.util.module_from_spec(spec);spec.loader.exec_module(hunt)
    evidence=hunt.read_evidence(seed,profile,result.returncode);evidence['wall_seconds']=time.time()-started
    (profile/'evidence.json').write_text(json.dumps(evidence,indent=2)+'\n')
    print('QA30 SHUTDOWN REPLAY:',seed,'exit='+str(result.returncode),'outcome='+str((evidence.get('finish') or {}).get('outcome')),flush=True)
    return dict(seed=seed,returncode=result.returncode,argv=command,evidence=str(profile/'evidence.json'))

def main():
    campaign=json.loads((EVIDENCE/'campaign-default/campaign.json').read_text())
    warning=next((run['seed'] for run in sorted(campaign['runs'],key=lambda row:row['seed']) if any('ObjectDB instances leaked' in finding['error'] for finding in run['findings'])),None)
    if warning is None:
        print('QA30 SHUTDOWN REPLAY: not applicable; no warning',flush=True);return 0
    seeds=list(dict.fromkeys((CONTROL_SEED,warning)))
    with ThreadPoolExecutor(max_workers=REPLAY_PROCESSES) as pool:records=list(pool.map(execute,seeds))
    (EVIDENCE/'shutdown-replays.json').write_text(json.dumps(records,indent=2)+'\n')
    return 0

if __name__=='__main__':raise SystemExit(main())
