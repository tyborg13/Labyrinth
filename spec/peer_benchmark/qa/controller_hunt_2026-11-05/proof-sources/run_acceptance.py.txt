"""Run six physical-input campaigns, at most two Godot processes at once."""
import concurrent.futures
import json
from pathlib import Path
import subprocess
import time
ROOT = Path('/Users/borgerding/workspace/Labyrinth.worktrees/qa-51')
EVIDENCE = Path('/private/tmp/labyrinth_codex/qa-51/evidence/campaign-acceptance')
SEEDS = (4, 5, 6, 7, 8, 9)
PROCESS_LIMIT = 4
PROCESS_TIMEOUT_SECONDS = 3600
SCRIPT = '/private/tmp/labyrinth_codex/qa-51/tools/controller_hunt.gd'
def run(seed):
    output = EVIDENCE / str(seed)
    output.mkdir(parents=True, exist_ok=True)
    command = ['python3', 'tools/godot_task_runner.py', '--task-id', 'qa-51', '--timeout', str(PROCESS_TIMEOUT_SECONDS), '--stream', '--', 'godot', '--headless', '--path', '.', '--script', SCRIPT, '--', '--seed', str(seed), '--no-embrace', 'true' if seed == 4 else 'false', '--output-dir', str(output)]
    (output / 'command.json').write_text(json.dumps(command, indent=2)+'\n')
    start = time.monotonic()
    with (output / 'execution.log').open('w') as log:
        result = subprocess.run(command, cwd=ROOT, stdout=log, stderr=subprocess.STDOUT)
    record = dict(seed=seed, returncode=result.returncode, elapsed_seconds=time.monotonic()-start)
    (output / 'execution.json').write_text(json.dumps(record, indent=2)+'\n')
    print(json.dumps(record), flush=True)
    return record
if __name__ == '__main__':
    EVIDENCE.mkdir(parents=True, exist_ok=True)
    with concurrent.futures.ThreadPoolExecutor(max_workers=PROCESS_LIMIT) as pool:
        rows = list(pool.map(run, SEEDS))
    (EVIDENCE / 'execution.json').write_text(json.dumps(rows, indent=2)+'\n')
