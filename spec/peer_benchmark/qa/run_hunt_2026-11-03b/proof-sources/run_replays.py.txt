"""Run required read-only census replays after campaign and Python completion."""
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
import json
import os
import subprocess
import sys
import time

PROJECT = Path('/Users/borgerding/workspace/Labyrinth.worktrees/qa-25')
EVIDENCE = Path('/private/tmp/labyrinth_codex/qa-25')
LAUNCHER = EVIDENCE/'run_hunt_task.py'
CENSUS = EVIDENCE/'run_hunt_census.gd'
COORDINATOR_POLL_SECONDS = 1
REPLAY_CASES = [('replay-default-seed-12',12,[]),('replay-no-embrace-seed-2',2,['--no-embrace'])]

def wait_for_campaigns():
    completion = EVIDENCE/'python-default-execution.json'
    coordinator = json.loads((EVIDENCE/'coordinator-pid.json').read_text())['pid']
    denied_reported = False
    while not completion.exists():
        try:os.kill(coordinator,0)
        except PermissionError:
            if not denied_reported: print('QA25 CENSUS WAIT: signal-0 liveness denied; waiting on coordinator completion record.',flush=True)
            denied_reported = True
        except ProcessLookupError:raise RuntimeError('Campaign coordinator stopped before Python completion; preserve evidence and resume pending work.')
        time.sleep(COORDINATOR_POLL_SECONDS)
    if json.loads(completion.read_text())['returncode']:
        raise RuntimeError('Python default group failed; inspect its recorded results before census.')

def run(case):
    name,seed,options = case
    output = EVIDENCE/name
    execution = EVIDENCE/(name+'-execution.json')
    if execution.exists():
        previous = json.loads(execution.read_text())
        if previous['returncode']==0 and (output/'campaign.json').exists():
            print('QA25 CENSUS RETAINED:',name,flush=True)
            return previous
    if output.exists() and any(output.iterdir()):raise RuntimeError('Partial replay already exists; preserve it and use a fresh retry directory: '+str(output))
    command = [sys.executable,str(LAUNCHER),'--runs','1','--processes','1','--seed',str(seed),'--script',str(CENSUS),'--output-dir',str(output),*options]
    started = time.time()
    print('QA25 CENSUS START:',name,flush=True)
    with (EVIDENCE/(name+'.log')).open('w') as log:
        result = subprocess.run(command,cwd=PROJECT,env=dict(os.environ,LABYRINTH_CEL_LOOK='off'),stdout=log,stderr=subprocess.STDOUT)
    record = dict(name=name,seed=seed,argv=command,started=started,wall_seconds=time.time()-started,returncode=result.returncode)
    execution.write_text(json.dumps(record,indent=2)+'\n')
    print('QA25 CENSUS END:',name,'exit='+str(result.returncode),flush=True)
    return record

def main():
    wait_for_campaigns()
    with ThreadPoolExecutor(max_workers=len(REPLAY_CASES)) as pool:
        records = list(pool.map(run,REPLAY_CASES))
    (EVIDENCE/'census-executions.json').write_text(json.dumps(records,indent=2)+'\n')
    return int(any(record['returncode'] for record in records))

if __name__=='__main__':sys.exit(main())
