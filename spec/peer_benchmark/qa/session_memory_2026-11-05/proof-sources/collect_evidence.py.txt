from pathlib import Path
import hashlib,json,re,shutil,subprocess
ROOT=Path('/Users/borgerding/workspace/Labyrinth.worktrees/qa-53')
TASK=Path('/private/tmp/labyrinth_codex/qa-53')
REPORT=ROOT/'spec/peer_benchmark/qa/session_memory_2026-11-05'
METRICS=['static_memory','video_memory','texture_memory','object_count','node_count','orphan_count','resource_count']
index=[]
for phase,label in [('before','before'),('after','after-clean')]:
 directory=TASK/label;session=json.loads((directory/'session.json').read_text());compact=json.loads(json.dumps(session))
 table=['| Boundary | Static bytes | Video bytes* | Texture bytes* | Objects | Nodes | Orphans | Resources |','|---|---:|---:|---:|---:|---:|---:|---:|']
 classes=['# '+phase+' native class breakdowns','','Counts include two transient bridge objects; monitor samples precede those objects.','']
 for b,c in zip(session['boundaries'],compact['boundaries']):
  table.append('| '+' | '.join([b['label'],*[str(int(b[k])) for k in METRICS]])+' |')
  classes += ['## '+b['label'],'','| Class | Instances |','|---|---:|']+[f"| {r['class']} | {int(r['count'])} |" for r in b['top_20']]+['']
  for cache in c.get('asset_cache_owners',{}).values():cache.pop('references',None)
  raw=directory/(b['label']+'-objects.json');c['object_census_path']=str(raw)
  index.append({'phase':phase,'boundary':b['label'],'objects':str(raw),'sha256':hashlib.sha256(raw.read_bytes()).hexdigest()})
 (REPORT/(phase+'-boundaries.md')).write_text('\n'.join(table)+'\n\n*Headless GPU zeros are unavailable measurements.\n')
 (REPORT/(phase+'-top-20.md')).write_text('\n'.join(classes))
 (REPORT/(phase+'-boundaries.json')).write_text(json.dumps(compact,indent=2)+'\n')
 shutil.copyfile(directory/'execution.json',REPORT/(phase+'-execution.json'))
 exit_markers=[line for line in (directory/'engine.log').read_text().splitlines() if any(x in line for x in ['ObjectDB','resources still in use','Leaked instance:','Resource still in use:','Orphan StringName:','unclaimed string names'])]
 (REPORT/(phase+'-exit.txt')).write_text('\n'.join(exit_markers)+'\n')
shutil.copyfile(TASK/'after-clean/semantic-integrity.json',REPORT/'semantic-integrity.json')
shutil.copyfile(TASK/'after-clean/integrity-comparison.json',REPORT/'strict-integrity-comparison.json')
# Exact stale-ID membership, independent of reference-graph inference.
session=json.loads((TASK/'after-clean/session.json').read_text());audit=[]
for b in session['boundaries']:
 if not re.fullmatch(r'run-[1-5]-menu',b['label']):continue
 objects={r['id'] for r in json.loads((TASK/'after-clean'/(b['label']+'-objects.json')).read_text())}
 keys=b['asset_cache_owners']['_texture_used_rect_cache']['keys']
 dead=[k for k in keys if (m:=re.search(r'instance\|(-?\d+)',k)) and m.group(1) not in objects]
 audit.append({'boundary':b['label'],'cache_entries':len(keys),'dead_instance_keys':dead,'dead_count':len(dead)})
(REPORT/'stale-bounds-key-audit.json').write_text(json.dumps(audit,indent=2)+'\n')
for path in (TASK/'tools').iterdir():
 if path.suffix in {'.gd','.py','.cpp','.gdextension'} and path.name not in {'empty.gd','census_smoke.gd','census_smoke_pure.gd'}:
  shutil.copyfile(path,REPORT/'proof-sources'/(path.name+'.txt'))
check=subprocess.run(['git','diff','--exit-code','--','scripts','scenes','data','assets','fonts','shaders','tests','tools','project.godot','spec/analytics.md'],cwd=ROOT,capture_output=True,text=True)
(REPORT/'source-check.json').write_text(json.dumps({'head':subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip(),'production_diff_exit':check.returncode,'production_diff':check.stdout,'note':'Only the owned report directory is added to the worktree. No production source, permanent test, asset, data value or registry changed.'},indent=2)+'\n')
(REPORT/'evidence-index.json').write_text(json.dumps({'raw_population_files':index,'raw_logs':[str(TASK/l/'engine.log') for l in ['before','after','after-clean']],'tools':str(TASK/'tools'),'native_captures':[]},indent=2)+'\n')
print('QA53 EVIDENCE: 22 boundary summaries, 22 top-20 breakdowns, stale-ID audit, semantic/strict reports and unchanged-source check written')
