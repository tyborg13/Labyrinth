extends SceneTree
const SETTLE_FRAMES: int = 4
var shader_case: String = "empty"
func _initialize() -> void:
	for arg: String in OS.get_cmdline_user_args():
		if arg.begins_with("--case="): shader_case = arg.trim_prefix("--case=")
	call_deferred("_run")
func _load_shader(path: String) -> void:
	var shader: Shader = load(path)
	var material := ShaderMaterial.new()
	material.shader = shader
	shader.get_shader_uniform_list()
func _run() -> void:
	if shader_case in ["quantize", "both"]: _load_shader("res://shaders/pixel_fx_quantize.gdshader")
	if shader_case in ["lines", "both"]: _load_shader("res://shaders/pixel_fx_lines.gdshader")
	for frame: int in range(SETTLE_FRAMES): await process_frame
	print("QA53 SHADER EXIT: " + shader_case)
	quit()
