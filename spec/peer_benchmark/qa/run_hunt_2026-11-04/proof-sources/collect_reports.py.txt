"""Retain original campaign coverage without inflating it with replay runs."""
from collections import Counter
from pathlib import Path
import json
import shutil

PROJECT = Path('/Users/borgerding/workspace/Labyrinth.worktrees/qa-43')
EVIDENCE = Path('/private/tmp/labyrinth_codex/qa-43')
REPORT = PROJECT / 'spec/peer_benchmark/qa/run_hunt_2026-11-04'


def main():
    coverage = {}
    for name in ('default', 'no-embrace'):
        directory = EVIDENCE / ('campaign-' + name)
        campaign = json.loads((directory / 'campaign.json').read_text())
        rooms = Counter()
        sections = Counter()
        outcomes = Counter()
        final_dragons = 0
        bosses = Counter()
        final_dragon_ids = Counter()
        decisions = 0
        seeds = []
        for run in sorted(campaign['runs'], key=lambda item: item['seed']):
            finish = run.get('finish') or {}
            outcome = finish.get('outcome', 'process_failure')
            outcomes[outcome] += 1
            decisions += finish.get('decisions', 0)
            for room in run['rooms']:
                rooms[room['room_kind']] += 1
                if room['room_kind'] == 'boss':
                    boss_id = room.get('room_metadata', {}).get('boss_id', 'unavailable')
                    bosses[boss_id] += 1
                    if room['section'] == 5:
                        final_dragons += 1
                        final_dragon_ids[boss_id] += 1
            sections.update(set(room['section'] for room in run['rooms']))
            seeds.append(dict(seed=run['seed'], outcome=outcome,
                              decisions=finish.get('decisions'), returncode=run['returncode'],
                              sections=sorted(set(room['section'] for room in run['rooms'])),
                              rooms=len(run['rooms']), wall_seconds=run['wall_seconds']))
        coverage[name] = dict(runs_started=campaign['runs_started'], runs_ended=len(seeds),
                              completions=sum(outcomes[key] for key in ('victory', 'defeat', 'embrace')),
                              outcomes=dict(outcomes), rooms=dict(rooms), sections=dict(sections),
                              room_entries=sum(rooms.values()), bosses=dict(bosses),
                              final_dragon_ids=dict(final_dragon_ids), final_dragon_room_entries=final_dragons,
                              final_dragon_victories=outcomes['victory'], decisions=decisions,
                              wall_seconds=campaign['wall_seconds'], options=campaign['options'], seeds=seeds)
        for filename in ('report.md', 'findings.json', 'provenance.json'):
            shutil.copyfile(directory / filename, REPORT / (name + '-' + filename))
        shutil.copyfile(EVIDENCE / ('campaign-' + name + '-execution.json'),
                        REPORT / ('campaign-' + name + '-execution.json'))
    (REPORT / 'coverage.json').write_text(json.dumps(coverage, indent=2) + '\n')
    for source in EVIDENCE.iterdir():
        if source.suffix in ('.py', '.gd'):
            shutil.copyfile(source, REPORT / 'proof-sources' / (source.name + '.txt'))
    print('QA43 CAMPAIGN COVERAGE:', json.dumps({name: {key: row[key] for key in
          ('runs_ended', 'outcomes', 'room_entries', 'decisions', 'final_dragon_room_entries', 'final_dragon_victories')}
          for name, row in coverage.items()}))


if __name__ == '__main__':
    main()
