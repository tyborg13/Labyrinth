#!/usr/bin/env python3
"""Full original-driver follow-ups on the final renderer patch; two slots."""
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
import hashlib
import json
import os
import subprocess
import sys
import time

PROJECT = Path('/Users/borgerding/workspace/Labyrinth.worktrees/qa-23')
EVIDENCE = Path('/private/tmp/labyrinth_codex/qa-23')
FOLLOWUP_SEEDS = (3, 4)
MAX_ACTIVE_REPLAYS = 2

def run(seed):
    name = 'replay-no-embrace-seed-' + str(seed) + '-fixed'
    command = [sys.executable, str(EVIDENCE / 'run_hunt_task.py'), '--seed', str(seed),
        '--runs', '1', '--processes', '1', '--no-embrace', '--output-dir', str(EVIDENCE/name)]
    record = dict(seed=seed, argv=command, started=time.time(),
        product_sha256=hashlib.sha256((PROJECT/'scripts/combat_art_treatment.gd').read_bytes()).hexdigest(),
        driver='tools/run_hunt.gd (unchanged original)', output=str(EVIDENCE/name))
    print('QA23 FIXED REPLAY START: seed=' + str(seed), flush=True)
    with (EVIDENCE/(name+'.log')).open('w') as log:
        result = subprocess.run(command, cwd=PROJECT, env=dict(os.environ, LABYRINTH_CEL_LOOK='off'),
            stdout=log, stderr=subprocess.STDOUT)
    record.update(returncode=result.returncode, wall_seconds=time.time()-record['started'])
    (EVIDENCE/(name+'-execution.json')).write_text(json.dumps(record,indent=2)+'\n')
    print('QA23 FIXED REPLAY END: seed=' + str(seed) + ' exit=' + str(result.returncode), flush=True)
    return record

def main():
    with ThreadPoolExecutor(max_workers=MAX_ACTIVE_REPLAYS) as pool:
        results = list(pool.map(run,FOLLOWUP_SEEDS))
    (EVIDENCE/'postfix-replay-results.json').write_text(json.dumps(results,indent=2)+'\n')
    print('QA23 FIXED REPLAYS: completed=' + str(len(results)), flush=True)
    return int(any(r['returncode'] for r in results))

if __name__ == '__main__':
    sys.exit(main())
