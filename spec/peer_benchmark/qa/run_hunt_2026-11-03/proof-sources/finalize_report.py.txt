def read(path):
    return json.loads(path.read_text())

def category(error):
    if 'ret != noErr' in error: return 'platform certificate diagnostic'
    if "'actor_key'" in error: return 'protected contact payload'
    if 'native Godot exit 0; runner detected script errors' in error: return 'protected contact runner status'
    if 'Invalid polygon data, triangulation failed.' in error: return 'fixed polygon rejection'
    if 'Fight exploration budget reached 400 decisions' in error: return 'hunt exploration budget'
    if 'tree_nodes grew more than 25%' in error: return 'aggregate tree-growth flag'
    raise RuntimeError('Untriaged finding: '+error)

def load_replays():
    rows=[]
    for campaign,names in REPLAYS.items():
        for name in names:
            paths=list((EVIDENCE/name).glob('seed-*/evidence.json'))
            if len(paths)!=1: raise RuntimeError('Incomplete or ambiguous replay: '+name)
            e=read(paths[0])
            if not e.get('finish'): raise RuntimeError('Replay lacks finish evidence: '+name)
            for finding in e['findings']: category(finding['error'])
            row=dict(campaign=campaign,name=name,evidence=str(paths[0]),child_returncode=e['returncode'],
                outcome=e['finish']['outcome'],decisions=e['finish']['decisions'],sections=e['finish']['sections'],
                findings=[dict(category=category(f['error']),error=f['error'],location=f['location']) for f in e['findings']])
            if name.endswith('-fixed') and any(f['category']=='fixed polygon rejection' for f in row['findings']):
                raise RuntimeError('Fixed replay still has polygon error: '+name)
            rows.append(row)
    return rows

def census_rows():
    rows=[]
    for names in REPLAYS.values():
        for name in names:
            paths=list((EVIDENCE/name).glob('seed-*/trace.jsonl'))
            samples=[]
            for line in paths[0].read_text().splitlines():
                e=json.loads(line)
                if e['event']=='qa23_census': samples.append(e['data'])
            a=next((s for s in samples if s['context']['section']==0),None)
            b=next((s for s in samples if s['context']['section']==5),None)
            if a and b:
                hosts={k:b['hosts'].get(k,0)-a['hosts'].get(k,0) for k in set(a['hosts'])|set(b['hosts'])}
                rows.append(dict(replay=name,baseline=a['context'],section_5=b['context'],
                    pools_before=a['pools'],pools_after=b['pools'],
                    largest_host_deltas=sorted(hosts.items(),key=lambda x:x[1],reverse=True)[:8]))
    return rows
