from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
import hashlib,json,os,subprocess,sys,time
PROJECT=Path('/Users/borgerding/workspace/Labyrinth.worktrees/qa-45')
EVIDENCE=Path('/private/tmp/labyrinth_codex/qa-45')
REPLAY_PROCESSES=2
OBSERVER_CONFIRM_STOP=90

def run(name,seed,no_embrace,stop_after=0,verbose=False,script_override=None):
 directory=EVIDENCE/name
 launcher=EVIDENCE/('run_hunt_task_verbose.py' if verbose else 'run_hunt_task.py')
 script=script_override or ('tools/run_hunt.gd' if verbose else str(EVIDENCE/'run_hunt_replay.gd'))
 command=[sys.executable,str(launcher),'--seed',str(seed),'--runs','1','--processes','1','--script',script,'--output-dir',str(directory)]
 if no_embrace:command+=['--no-embrace']
 if stop_after:command+=['--stop-after',str(stop_after)]
 started=time.time();print('QA45 REPLAY START:',name,flush=True)
 with (EVIDENCE/(name+'.log')).open('w') as log:
  result=subprocess.run(command,cwd=PROJECT,env=dict(os.environ,LABYRINTH_CEL_LOOK='off'),stdout=log,stderr=subprocess.STDOUT)
 record=dict(name=name,seed=seed,no_embrace=no_embrace,argv=command,output_dir=str(directory),returncode=result.returncode,started=started,wall_seconds=time.time()-started)
 (EVIDENCE/(name+'-execution.json')).write_text(json.dumps(record,indent=2)+'\n')
 print('QA45 REPLAY END:',name,'exit='+str(result.returncode),flush=True)
 return record

def main():
 # Recheck the exact prefix containing the observer's first freed assignment.
 validation_path=EVIDENCE/'observer-confirm-validation.json'
 if validation_path.exists():
  validation=json.loads(validation_path.read_text())
  current={str(path):hashlib.sha256(path.read_bytes()).hexdigest() for path in [EVIDENCE/'run_hunt_replay.gd',EVIDENCE/'run_hunt_observed.gd',PROJECT/'tools/run_hunt.gd']}
  if validation.get('status')!='PASS' or validation.get('source_sha256')!=current:
   raise RuntimeError('Short confirmation is not valid for current probe inputs; rerun it with fresh output')
  confirm=json.loads((EVIDENCE/'observer-confirm-executions.json').read_text())[0]
  print('QA45 OBSERVER CONFIRM: reuse verified 90-decision prefix (source hashes match)',flush=True)
 else:
  confirm=run('observer-confirm-default-4',4,False,OBSERVER_CONFIRM_STOP)
  (EVIDENCE/'observer-confirm-executions.json').write_text(json.dumps([confirm],indent=2)+'\n')
 candidates=[]
 shutdown_candidates=[]
 for campaign in ['default','no-embrace']:
  data=json.loads((EVIDENCE/('campaign-'+campaign)/'campaign.json').read_text())
  shutdown_candidates.extend((campaign,row['seed']) for row in sorted(data['runs'],key=lambda row:row['seed']) if any('ObjectDB instances leaked at exit' in finding['error'] for finding in row['findings']))
  matching=[row for row in sorted(data['runs'],key=lambda row:row['seed']) if any(room['section']==5 for room in row['rooms'])]
  candidates.extend((campaign,row['seed']) for row in matching)
 selected=[]
 for campaign in ['default','no-embrace']:
  match=next((row for row in candidates if row[0]==campaign),None)
  if match:selected.append(match)
 for row in candidates:
  if len(selected)>=2:break
  if row not in selected:selected.append(row)
 if len(selected)<2:raise RuntimeError('Need two observed section-5 runs; retain coverage gap for owner')
 with ThreadPoolExecutor(max_workers=REPLAY_PROCESSES) as pool:
  futures=[pool.submit(run,'census-'+campaign+'-'+str(seed),seed,campaign=='no-embrace') for campaign,seed in selected]
  shutdown_futures=[pool.submit(run,'shutdown-'+campaign+'-'+str(seed),seed,campaign=='no-embrace',0,True) for campaign,seed in shutdown_candidates]
  results=[future.result() for future in futures]
  shutdown_results=[future.result() for future in shutdown_futures]
 (EVIDENCE/'census-executions.json').write_text(json.dumps(results,indent=2)+'\n')
 (EVIDENCE/'shutdown-executions.json').write_text(json.dumps(shutdown_results,indent=2)+'\n')
 return subprocess.call([sys.executable,str(EVIDENCE/'collect_census.py')],cwd=PROJECT)

if __name__=='__main__':raise SystemExit(main())
